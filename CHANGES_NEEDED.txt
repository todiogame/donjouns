    onlyBotsRemain() {
        // Check if game is in active phase
        const isGamePhase = typeof this.phase === "string" && this.phase.includes("GAME");
        if (!isGamePhase) {
            return false;
        }
        // Check if all active players (in dungeon) are bots
        const playersInDungeon = this.players.filter(p => p.inDungeon());
        if (playersInDungeon.length === 0) {
            return false;
        }
        return playersInDungeon.every(p => p.isBot);
    }
